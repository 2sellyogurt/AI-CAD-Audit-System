## 一、现状分析

### 1.1 当前系统实际能力（基于代码）

| 模块 | 实际功能 | 实际输入 | 实际输出 | 局限性 |
|------|---------|---------|---------|--------|
| `tools_extract_dxf.py` | 提取DXF文本 | DXF文件路径 | `{raw_texts, elevations, metadata}` | 只提取TEXT/MTEXT，不提取尺寸标注 |
| `rule_checker.py` | 规则合规检查 | `unified.json`+规则库 | `{rule_results, rule_stats}` | 262条规则=关键词搜索，无数值比对 |
| `elevation_calc.py` | 标高净高计算 | `unified.json` | `{functional_net_height, beam_dimensions}` | 净高=max(梁高,板厚)，非相加 |
| `collision_detect.py` | 碰撞检测 | `unified.json` | `{tray_collisions, beam_interference}` | 文本匹配，非几何计算 |
| `cross_discipline.py` | 跨专业比对 | `unified.json` | `{cross_results, deep_cross_results}` | 硬编码3条规则做数值比对 |
| `engineering_analysis.py` | 工程分析 | `unified.json` | `{constructability_score, ...}` | 成本用硬编码单价 |
| `report_generator_v50.py` | 报告生成 | 所有分析结果 | 3份MD报告 | 报告截断30/20条 |
| `meeting_list.py` | 会审清单 | 所有分析结果 | 会审问题清单MD | 梁开洞是空实现 |

### 1.2 行业知识现状（基于6份专监文档）

| 文档 | 可转化检查项 | 当前系统覆盖 | 差距 |
|------|------------|-------------|------|
| ①岗位能力与法规体系 | 35条 | 0条 | 导管交叉/桥架接地/防雷焊接等 |
| ②建筑类型与验收实战 | 35条 | 0条 | GB 55038-2025住宅新规/医疗专项 |
| ③材料阀门与系统调试 | 43条 | 0条 | 桥架板厚/风管板厚/管壁厚等 |
| ④甲方视角与质监验收 | 30条 | 0条 | A签扫码/防火封堵/保温检查 |
| ⑤具体数值与测试试验 | 74条 | 0条 | 绝缘电阻/水压试验/启动时间等 |
| ⑥管线综合碰撞检测 | 40条 | 部分覆盖 | 间距标准/检修空间/抗震支吊架 |
| **合计** | **280条** | **0条** | **全需新增** |

```python
                    "value": self._get_dimension_value(entity),
                    "unit": "mm",
                    "label": self._get_dimension_label(entity),
                    "text": entity.dxf.text,
                    "measurement": entity.measurement,
                    "layer": entity.dxf.layer
                }

    def _extract_block_attributes(self, msp):
```

```python
                params_by_discipline, "建筑高度")
        }
        
        # 4. 建立参数字典（唯一真相源）
        anchor_dict = self._build_anchor_dict(
            params_by_discipline, cross_validation)
        
        return anchor_dict
    
    def _validate_cross_disc(self, params_by_disc, param_name):
        """跨专业验证参数一致性"""
        values = {}
        for disc, params in params_by_disc.items():
            extracted = self._extract_param_value(params, param_name)
            if extracted:
                values[disc] = extracted
        
        if len(set(values.values())) == 1:
            status = "consistent"
        elif len(values) == 0:
            status = "not_found"
        else:
            status = "inconsistent"
        
        return {
            "values": values,
            "status": status,
            "reference_value": list(values.values())[0] if values else None
        }
```

```python
class RuleFilter:
    """规则预筛选器"""
    
    def filter(self, all_rules, project_anchor, review_phase):
        """
        按项目特征过滤规则，减少无效检查
        """
        filtered = []
        
        for rule in all_rules:
            # 1. 建筑类型过滤
            if not self._is_applicable_type(rule, project_anchor):
                continue
            
            # 2. 专业过滤
            if not self._is_applicable_discipline(rule, project_anchor):
                continue
            
            # 3. 阶段过滤
            if not self._is_applicable_phase(rule, review_phase):
                continue
            
            # 4. 规则优先级排序
            priority = self._calculate_priority(rule, project_anchor)
            rule['__priority'] = priority
            filtered.append(rule)
        
        # 按优先级排序：严重>重要>一般
        return sorted(filtered, key=lambda r: (
            {'严重': 0, '重要': 1, '一般': 2, '提示': 3}.get(r['severity'], 4),
            r['__priority']
        ))
    
    def _is_applicable_type(self, rule, anchor):
        """检查规则是否适用于当前建筑类型"""
        applicable_types = rule.get('applicable_types', ['all'])
        if 'all' in applicable_types:
            return True
        return anchor['building_type'] in applicable_types
```

```python
# engine/value_compare_checker.py

class ValueCompareChecker:
    """数值比对检查器"""
    
    def check(self, rule, unified_file):
        """
        对数值类规则进行提取和比对
        """
        extraction_config = rule['value_extraction']
        validation_config = rule['validation']
        
        # 1. 从文本中提取数值
        extracted_values = self._extract_values(
            unified_file['text_content']['raw_texts'],
            extraction_config
        )
        
        # 2. 从尺寸标注中提取数值（新增）
        if 'dimensions' in unified_file['text_content']:
            extracted_values.extend(
                self._extract_from_dimensions(
                    unified_file['text_content']['dimensions'],
                    extraction_config
                )
            )
        
        # 3. 数值比对验证
        results = []
        for value_set in extracted_values:
            validation_result = self._validate_value(value_set, validation_config)
            results.append({
                "extracted": value_set,
                "validation": validation_result
            })
        
        return self._aggregate_results(results, rule)
    
    def _validate_value(self, value_set, config):
        """按条件验证数值"""
        validation_type = config['type']
        
        if validation_type == 'conditional':
            # 条件验证：宽<300→厚≥1.0，300≤宽<500→厚≥1.2...
            conditions = config['conditions']
            for cond in conditions:
                if self._matches_condition(value_set, cond):
                    required = self._parse_requirement(cond['thickness'])
                    actual = value_set.get('thickness', 0)
                    return {
                        "pass": actual >= required['value'],
                        "actual": actual,
                        "required": required
                    }
        
        elif validation_type == 'range':
            # 范围验证：100≤值≤200
            min_val = config.get('min_value', float('-inf'))
            max_val = config.get('max_value', float('inf'))
            return {
                "pass": min_val <= value_set['value'] <= max_val,
                "actual": value_set['value'],
                "required": {"min": min_val, "max": max_val}
```

```python
# report/test_checklist_generator.py

class TestChecklistGenerator:
    """测试验收清单生成器"""
    
    def generate(self, project_info, applicable_rules):
        """
        根据适用规则生成测试验收清单
        """
        checklist = {
            "project_info": project_info,
            "summary": {"total_tests": 0, "by_discipline": {}},
            "checklist": []
        }
        
        # 筛选测试类规则
        test_rules = [r for r in applicable_rules 
                      if r.get('category') == '测试验收']
        
        # 按专业分组
        by_discipline = {}
        for rule in test_rules:
            disc = rule['discipline']
            if disc not in by_discipline:
                by_discipline[disc] = []
            by_discipline[disc].append(rule)
        
        # 生成清单项
        for disc, rules in by_discipline.items():
            for idx, rule in enumerate(rules):
                checklist_item = {
                    "id": f"TEST-{disc[:2].upper()}-{idx+1:03d}",
                    "test_item": rule['name'],
                    "discipline": disc,
                    "applicable_location": self._infer_location(rule, project_info),
                    "value_requirement": rule.get('value_requirement', {}),
                    "test_method": rule.get('test_method', {}),
                    "acceptance_criteria": {
                        "criterion": rule.get('acceptance_criteria', ''),
                        "description": rule.get('description', '')
                    },
                    "code_reference": rule.get('code_reference', ''),
                    "severity": rule.get('severity', '一般'),
                    "is_critical": rule.get('severity') == '严重',
                    "report_required": rule.get('report_required', False),
                    "common_defects": rule.get('common_defects', []),
                    "prerequisites": rule.get('prerequisites', []),
                    "safety_precautions": rule.get('safety_precautions', [])
                }
                checklist['checklist'].append(checklist_item)
        
        # 统计
        checklist['summary']['total_tests'] = len(checklist['checklist'])
        checklist['summary']['by_discipline'] = {
            disc: len(rules) 
            for disc, rules in by_discipline.items()
        }
        checklist['summary']['critical_tests'] = sum(
            1 for item in checklist['checklist'] if item['is_critical']
        )
        
        return checklist
```

### 5.2 缓存设计

```python
# cache/design.py

class ReviewCache:
    """
    审查结果缓存
    支持增量计算：只重新计算变更文件的结果
    """
    
    def __init__(self, cache_dir='./cache'):
        self.cache_dir = cache_dir
        self.file_hash_db = os.path.join(cache_dir, 'file_hashes.json')
        self.results_cache = os.path.join(cache_dir, 'results.json')
        self.config_cache = os.path.join(cache_dir, 'config.json')
    
    def detect_changes(self, dxf_files):
        """检测哪些文件变更了"""
        current_hashes = {}
        changes = {"added": [], "modified": [], "deleted": []}
        
        # 读取已有哈希
        old_hashes = self._load_hashes()
        
        for f in dxf_files:
            h = self._compute_hash(f)
            current_hashes[f] = h
            if f not in old_hashes:
                changes["added"].append(f)
            elif h != old_hashes[f]:
                changes["modified"].append(f)
        
        # 检测删除
        for f in old_hashes:
            if f not in current_hashes:
                changes["deleted"].append(f)
        
        # 保存新哈希
        self._save_hashes(current_hashes)
        
        return changes
    
    def get_cached_result(self, file_key):
        """获取缓存的审查结果"""
        cache = self._load_results()
        return cache.get(file_key)
    
    def update_cache(self, file_key, result):
        """更新单个文件的审查结果"""
        cache = self._load_results()
        cache[file_key] = result
        self._save_results(cache)
    
    def get_incremental_work(self, dxf_files, cached_files):
        """
        计算增量工作：只返回需要重新处理的文件
        """
        changes = self.detect_changes(dxf_files)
        
        incremental_files = set()
        incremental_files.update(changes["added"])
        incremental_files.update(changes["modified"])
        
        # 如果修改了跨专业依赖的文件，也需要重新检查
        for f in changes["modified"]:
            deps = self._get_dependent_files(f, cached_files)
            incremental_files.update(deps)
        
        return list(incremental_files)
```
